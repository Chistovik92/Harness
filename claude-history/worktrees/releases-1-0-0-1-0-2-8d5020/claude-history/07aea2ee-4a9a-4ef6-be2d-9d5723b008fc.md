+/**
+ * Runs local Hugging Face models (transformers.js) in a worker thread and
+ * exposes the result as an async generator of chat chunks.
+ */
+
+import { Worker } from 'node:worker_threads'
+import type { ChatChunk } from '../renderer/lib/providers'
+
+export type LocalRunOptions = {
+  workerPath: string
+  cacheDir: string
+  model: string
+  prompt?: string
+  temperature?: number
+  maxTokens?: number
+  mode?: 'generate' | 'download'
+  signal?: AbortSignal
+}
+
+const HF_MODEL_ID = /^[\w.-]+\/[\w.-]+$/
+
+export function isValidModelId(id: unknown): id is string {
+  return typeof id === 'string' && HF_MODEL_ID.test(id)
+}
+
+export async function* runLocalModel(opts: LocalRunOptions): AsyncGenerator<ChatChunk> {
+  if (!isValidModelId(opts.model)) {
+    yield { type: 'error', error: `Invalid model id: ${String(opts.model)}` }
+    return
+  }
+
+  const queue: ChatChunk[] = []
+  let finished = false
+  let wake: (() => void) | null = null
+  const notify = () => { wake?.(); wake = null }
+
+  const worker = new Worker(opts.workerPath, {
+    workerData: {
+      mode: opts.mode ?? 'generate',
+      model: opts.model,
+      prompt: opts.prompt ?? '',
+      temperature: opts.temperature ?? 0.7,
+      maxTokens: opts.maxTokens ?? 256,
+      cacheDir: opts.cacheDir,
+    },
+  })
+
+  worker.on('message', (msg: ChatChunk) => {
+    queue.push(msg)
+    if (msg.type === 'done' || msg.type === 'error') finished = true
+    notify()
+  })
+  worker.on('error', (err: Error) => {
+    queue.push({ type: 'error', error: err.message })
+    finished = true
+    notify()
+  })
+  worker.on('exit', () => { finished = true; notify() })
+
+  const onAbort = () => { void worker.terminate() }
+  opts.signal?.addEventListener('abort', onAbort, { once: true })
+
+  try {
+    while (true) {
+      const chunk = queue.shift()
+      if (chunk) {
+        yield chunk
+        if (chunk.type === 'done' || chunk.type === 'error') return
+        continue
+      }
+      if (finished) {
+        yield { type: 'done' }
+        return
+      }
+      await new Promise<void>((resolve) => { wake = resolve })
+    }
+  } finally {
+    opts.signal?.removeEventListener('abort', onAbort)
+    void worker.terminate()
+  }
+}

+/**
+ * AES-256-GCM helpers for the encrypted provider store.
+ *
+ * Payload format: `v1.<iv>.<ciphertext>.<authTag>` (all base64). The auth tag
+ * is mandatory: without it GCM cannot detect tampering and decryption fails.
+ */
+
+import { randomBytes, createCipheriv, createDecipheriv } from 'node:crypto'
+
+const ALGORITHM = 'aes-256-gcm'
+const KEY_BYTES = 32
+const IV_BYTES = 12
+const VERSION = 'v1'
+
+export function generateKey(): Buffer {
+  return randomBytes(KEY_BYTES)
+}
+
+export function isValidKey(key: Buffer): boolean {
+  return key.length === KEY_BYTES
+}
+
+export function encrypt(text: string, key: Buffer): string {
+  const iv = randomBytes(IV_BYTES)
+  const cipher = createCipheriv(ALGORITHM, key, iv)
+  const ciphertext = Buffer.concat([cipher.update(text, 'utf8'), cipher.final()])
+  const tag = cipher.getAuthTag()
+  return [VERSION, iv.toString('base64'), ciphertext.toString('base64'), tag.toString('base64')].join('.')
+}
+
+export function decrypt(payload: string, key: Buffer): string {
+  const [version, ivB64, ctB64, tagB64] = payload.trim().split('.')
+  if (version !== VERSION || !ivB64 || ctB64 === undefined || !tagB64) {
+    throw new Error('Unsupported or corrupted encrypted payload')
+  }
+  const decipher = createDecipheriv(ALGORITHM, key, Buffer.from(ivB64, 'base64'))
+  decipher.setAuthTag(Buffer.from(tagB64, 'base64'))
+  return Buffer.concat([decipher.update(Buffer.from(ctB64, 'base64')), decipher.final()]).toString('utf8')

+ * Minimal Server-Sent Events reader shared by all HTTP providers.
+ *
+ * Yields the payload of every `data:` line. Handles CRLF line endings,
+ * multi-chunk lines, a trailing line without a final newline and abort.
+ */

+export async function* readSSE(res: Response, signal?: AbortSignal): AsyncGenerator<string> {
+  if (!res.body) return
+  const reader = res.body.getReader()
+  const decoder = new TextDecoder()
+  let buffer = ''
+
+  const onAbort = () => { reader.cancel().catch(() => {}) }
+  signal?.addEventListener('abort', onAbort, { once: true })
+
+  try {
+    while (true) {
+      const { done, value } = await reader.read()
+      if (done) break
+      buffer += decoder.decode(value, { stream: true })
+      const lines = buffer.split(/\r?\n/)
+      buffer = lines.pop() ?? ''
+      for (const line of lines) {
+        const payload = parseDataLine(line)
+        if (payload !== null) yield payload

+    if (tail !== null) yield tail
+  } finally {

+  }
